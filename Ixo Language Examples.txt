"""
    Error handling

    Functions that can fail usually return T ! Error (T or error)

    Return type T ! Error means:
    - success → returns value of type T
    - failure → returns an error value (type Error or subtype)
    The ! branch is executed only on failure

    # In error branch:
    retry              # retry immediately
    retry 3            # limited retry (optional)
    retry 500ms        # delay before retry

    Examples:

    content = fs.read(path) ! err
        err.log_error("Read failed:")
        return null

    value = risky_calc ! e -> return Error.CalculationFailed(e)

    data = api.fetch ! err
        retry_count++
        if retry_count > 3 -> return Error.MaxRetries
        retry 500ms

        # Propagating (Passing the buck), using ?
        fn read_config
            file = fs.open("config")?   
            # If open fails, read_config returns the error NOW
            data = file.read?
            # If read fails, read_config returns the error NOW
            data.parse
"""

# Concurrency

# fork fn ...       → starts a lightweight concurrent task (non-blocking)
# flow[T]           → safe directed data pipe (send :> / receive <:)
# close(flow)       → signals no more values will be sent
# with tasks        → auto-awaits and cleans up all forked tasks inside

# Create flows (safe data pipes)
jobs    = flow[int](100)                     # buffered
results = flow[Result[str, Error]]           # unbuffered

# Send: value :> flow   (push right)
# Receive: value <: flow (pull left)

# Fork starts a lightweight concurrent task
fork fn worker(id: int, jobs: flow[int], results: flow[Result[str, Error]])
    for job <: jobs
        job.compute :> results

# Full example: concurrent workers
fn main
# Unless specified, the whole file is implicitely the main function.
    jobs    = flow[int](100)
    results = flow[Result[str, Error]](100)

    for i in 1..=8
        fork worker(i, jobs, results)

    # send work (arrow style)
    for j in 1..=1000 -> j :> jobs

    jobs.close

    # receive results
    for _ in 1..=1000
        result <: results
        result ! err -> err.log_error
        result.print

# Structured concurrency (auto-wait & cleanup)
with tasks
    for url in urls
        fork fetch(url) -> page
            page.process

    # block ends → all forked tasks are awaited automatically

# With arenas (zero-pause concurrency)
@frame
    fork update_particles(particle_pool)
    fork update_lights(light_cache)
    # both forked tasks share the @frame arena → no GC pauses

# Quick showcase – concurrent 3D particle update
fn update_particles
    @frame $jit
        positions = flow[vec]
        velocities = flow[vec]

        fork generate_particles -> p
            p :> positions

        fork apply_forces(positions) -> new_pos
            new_pos :> velocities

        for pos <: positions, vel <: velocities
            pos + vel * delta -> pos
            pos.print
